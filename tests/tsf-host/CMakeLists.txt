cmake_minimum_required(VERSION 3.25)
project(MsimeTsfHost LANGUAGES CXX)
add_executable(msime-tsf-host WIN32 main.cpp)
target_compile_features(msime-tsf-host PRIVATE cxx_std_17)
target_compile_definitions(msime-tsf-host PRIVATE UNICODE _UNICODE)
target_compile_options(msime-tsf-host PRIVATE /utf-8)
set(MSIME_TSF_HOST_STATE_PATH "" CACHE FILEPATH "Manual fixture export path; secret controls are never exported")
if(MSIME_TSF_HOST_STATE_PATH)
    file(TO_CMAKE_PATH "${MSIME_TSF_HOST_STATE_PATH}" MSIME_TSF_HOST_STATE_PATH_NORMALIZED)
    target_compile_definitions(msime-tsf-host PRIVATE "MSIME_TSF_HOST_STATE_PATH=L\"${MSIME_TSF_HOST_STATE_PATH_NORMALIZED}\"")
endif()
target_link_libraries(msime-tsf-host PRIVATE ole32 uuid user32 gdi32)
