cmake_minimum_required(VERSION 3.25)
project(NativeRecoveryProbe LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "Standalone integration probe" FORCE)
add_subdirectory("../../../ui" ui)
add_executable(native_recovery_probe probe.cpp)
target_compile_features(native_recovery_probe PRIVATE cxx_std_17)
target_compile_definitions(native_recovery_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(native_recovery_probe PRIVATE ../../src)
target_link_libraries(native_recovery_probe PRIVATE msimeui user32 gdi32 dwmapi ole32)
